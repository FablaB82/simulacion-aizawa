    <style>

    <div id="ui">

        Frecuencia (d): <span id="val">3.5</span><br>

        <input type="range" id="freq" min="1" max="7" step="0.1" value="3.5">

    </div>


    <script>

        let scene, camera, renderer, line, geometry;

        let x = 0.1, y = 0, z = 0;

        let points =;

        let params = { a: 0.95, b: 0.7, c: 0.6, d: 3.5, e: 0.25, f: 0.1 };

        const dt = 0.01;

        const maxPoints = 5000;


        init();

        animate();


        function init() {

            scene = new THREE.Scene();

            camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);

            renderer = new THREE.WebGLRenderer({ antialias: true });

            renderer.setSize(window.innerWidth, window.innerHeight);

            document.body.appendChild(renderer.domElement);


            geometry = new THREE.BufferGeometry();

            const material = new THREE.LineBasicMaterial({ color: 0x00ffcc });

            line = new THREE.Line(geometry, material);

            scene.add(line);


            camera.position.z = 50;


            document.getElementById('freq').addEventListener('input', (e) => {

                params.d = parseFloat(e.target.value);

                document.getElementById('val').innerText = params.d;

                // Reiniciar puntos al cambiar frecuencia para ver la transición limpia

                points =; 

                x = 0.1; y = 0; z = 0;

            });

        }


        function animate() {

            requestAnimationFrame(animate);


            // Ecuaciones de Aizawa 

            let dx = ((z - params.b) * x - params.d * y) * dt;

            let dy = (params.d * x + (z - params.b) * y) * dt;

            let dz = (params.c + params.a * z - Math.pow(z, 3)/3 - (x*x + y*y) * (1 + params.e * z) + params.f * z * Math.pow(x, 3)) * dt;


            x += dx; y += dy; z += dz;


            // Aquí podrías añadir el offset en Z para el modo "grabación"

            // let zOffset = points.length * 0.05; 

            points.push(new THREE.Vector3(x * 10, y * 10, z * 10));


            if (points.length > maxPoints) points.shift();


            geometry.setFromPoints(points);

            line.rotation.y += 0.01;

            renderer.render(scene, camera);

        }

    </script>
